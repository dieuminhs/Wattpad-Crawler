Purpose: When Wattpad rejects auth mid-job (cookie expired between archive start and
chapter N), abort immediately with a structured exception instead of:
- silently consuming all retry attempts on a deterministic failure (D-14), OR
- letting `raise_for_status()` raise a generic `httpx.HTTPStatusError` that doesn't carry

**Wattpad's actual unauth contract (verified manually 2026-05-03 via Plan 02-01 Task 1):**
unauthenticated requests to `/api/v3/users/wattpad/library?limit=1` return HTTP **400** with
JSON body `{"error_code":1018,"error_type":"PermissionDenied","message":"User not logged in"}`,
NOT 401/403/redirect. The branch must therefore detect THREE shapes:
1. `resp.status_code == 401`
2. `resp.status_code == 403`
3. `resp.status_code == 400` AND response body has `error_type:"PermissionDenied"` OR `error_code:1018`

Plain HTTP 400 responses (e.g., `error_type:"InvalidEndpoint"`, `error_code:1001`) MUST
fall through to existing handling — they are real client errors, not auth failures.

def test_get_does_not_retry_on_401(tmp_path):
    """AUTH-04 / D-14: first 401 fails immediately, no retries."""
    calls = {"n": 0}
    def handler(req):
        calls["n"] += 1

        with pytest.raises(AuthFailedError):

    finally:
        rlc.close()


def test_get_raises_on_403(tmp_path):
    """AUTH-04 / D-13: 403 triggers AuthFailedError just like 401."""
    calls = {"n": 0}
    def handler(req):
        calls["n"] += 1
        return httpx.Response(403)
    transport = httpx.MockTransport(handler)
    rlc = make_client(tmp_path, transport)
    try:
        with pytest.raises(AuthFailedError):
            rlc.get("https://www.wattpad.com/x", max_attempts=5)
        assert calls["n"] == 1
    finally:
        rlc.close()


def test_auth_failed_error_payload(tmp_path):
    """AUTH-04 / D-15: AuthFailedError carries status_code and url attributes."""
    transport = httpx.MockTransport(lambda req: httpx.Response(401))
    rlc = make_client(tmp_path, transport)
    try:
        with pytest.raises(AuthFailedError) as exc_info:
            rlc.get("https://www.wattpad.com/x")
        assert exc_info.value.status_code == 401
        assert exc_info.value.url == "https://www.wattpad.com/x"
    finally:
        rlc.close()


def test_get_raises_on_400_permission_denied(tmp_path):
    """AUTH-04: Wattpad's actual mid-job unauth signal is HTTP 400 + PermissionDenied,
    NOT 401/403. Verified manually 2026-05-03 against live API (Plan 02-01 Task 1).
    """
    calls = {"n": 0}
    def handler(req):
        calls["n"] += 1
        return httpx.Response(
            400,

                "error_code": 1018,
                "error_type": "PermissionDenied",
                "message": "User not logged in",

    transport = httpx.MockTransport(handler)
    rlc = make_client(tmp_path, transport)
    try:
        with pytest.raises(AuthFailedError) as exc_info:
            rlc.get("https://www.wattpad.com/x", max_attempts=5)
        assert exc_info.value.status_code == 400
        assert exc_info.value.url == "https://www.wattpad.com/x"
        assert calls["n"] == 1, f"Expected 1 call (no retry); got {calls['n']}"
    finally:
        rlc.close()


def test_get_does_not_intercept_400_invalid_endpoint(tmp_path):

        rlc.close()

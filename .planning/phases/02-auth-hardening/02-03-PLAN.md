```python
def main(argv: list[str] | None = None) -> int:
    args = build_parser().parse_args(argv)
    _setup_logging(args.verbose)
    cfg = load_config(args.output)
    client = RateLimitedClient(cfg)
    manifest = Manifest(cfg.output_dir).connect()
    try:
        try:
            if args.cmd == "story":
                _require_auth(client)
                sid = resolve_story_id(args.target)
                archive_story(cfg, client, manifest, sid)
            elif args.cmd == "url":
                _require_auth(client)
                sid = resolve_story_id(args.target)
                archive_story(cfg, client, manifest, sid)
            elif args.cmd == "library":
                _require_auth(client)
                from local_story_archive.api.user import fetch_library
                ids = fetch_library(client, args.user)
                archive_many(cfg, client, manifest, ids)
            elif args.cmd == "list":
                _require_auth(client)
                from local_story_archive.api.user import fetch_list_story_ids
                ids = fetch_list_story_ids(client, args.list_id)
                archive_many(cfg, client, manifest, ids)
            elif args.cmd == "status":
                # D-06: status reads local sqlite only — no validation.
                _print_status(manifest)
            elif args.cmd == "serve":
                # D-06: web /setup handles auth interactively.
                # serve owns its own client/manifest lifecycle inside JobRunner threads;
                # close the ones main() opened so we don't leak them.
                manifest.close()
                client.close()
                app = build_app(cfg)
                uvicorn.run(app, host=args.host, port=args.port, log_level="info")
                return 0
            return 0
        except AuthError as e:
            # D-08: print to stderr, no traceback noise, exit 2 (= misuse / config error).
            print(
                f"AuthError: {e}\n"
                f"Update your cookie via /setup or edit {cfg.output_dir}/_config.toml.",
                file=sys.stderr,
            )
            return 2
    finally:
        manifest.close()
        client.close()
```

NOTES on the structure:
- The OUTER try is the existing one with `finally: manifest.close(); client.close()` — keep
  it. We do NOT add `manifest.close()` inside the AuthError handler because the finally
  will run after the return.
- The INNER try/except catches AuthError from `_require_auth(client)`. validate_cookie may
  also raise httpx.RequestError on network failure — we let that propagate (D-02 says
  network errors are NOT auth errors; they should produce a different signal — for now,
  the existing finally will close client/manifest and the exception will bubble up to
  Python's default handler, which prints a traceback. That's acceptable per Phase 2 scope;
  improving network error UX is not in REQUIREMENTS).
- The remediation message format MUST contain the literal string "AuthError" (test asserts
  this) and the literal string "/setup" (test asserts this). Stick to the exact text shape.
- `cfg.output_dir` interpolation produces a Path repr like `wattpad-archive` (relative) —
  acceptable for a remediation hint.
